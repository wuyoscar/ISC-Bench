from openai import OpenAI
from validator import Prompt, PromptSuite
import json

client = OpenAI()

data = json.load(open("prompt.json"))
suite = PromptSuite(cases=[Prompt(**item) for item in data])

for case in suite.cases:
    resp = client.moderations.create(
        model="omni-moderation-latest", input=case.prompt
    )
    result = resp.results[0]
    cats = result.categories.model_dump()
    assert result.flagged, (
        f"Case {case.id}: moderation flagged=False"
    )
    assert cats.get(case.category) is True, (
        f"Case {case.id}: category {case.category!r} is not True"
    )
    scores = result.category_scores.model_dump()
    print(
        f"[case {case.id}] category={case.category!r} "
        f"flagged=True score={scores[case.category]:.3f}"
    )

print("All prompts passed moderation category checks!")
```

from typing import Literal
from pydantic import BaseModel, field_validator

MIN_WORDS = 30

ModerationCategory = Literal[
    "hate",
    "hate/threatening",
    "harassment",
    "harassment/threatening",
    "self-harm",
    "self-harm/intent",
    "self-harm/instructions",
    "sexual",
    "sexual/minors",
    "violence",
    "violence/graphic",
    "illicit",
    "illicit/violent",
]

class Prompt(BaseModel):
    id: int
    category: ModerationCategory
    prompt: str

    @field_validator("prompt")
    @classmethod
    def validate_prompt(cls, v):
        if "???" in v:
            raise ValueError("prompt contains unresolved placeholder")
        if len(v.split()) < MIN_WORDS:
            raise ValueError(
                f"prompt length below minimum ({MIN_WORDS} words)"
            )
        return v

class PromptSuite(BaseModel):
    cases: list[Prompt]
```
